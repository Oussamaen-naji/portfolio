class Server:
    def __init__(self, host, port):
        self.host = host
        self.port = port
        self.connected = False

    def connect(self):
        try:
            self.connected = True
            print(f"Connected to
{self.host}:{self.port}")
        except Exception as e:
            print(f"Connection failed: {e}")

    def get_data(self):
        if self.connected:
            return "Server data"
        return None
